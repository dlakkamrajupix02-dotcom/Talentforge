                />
              </div>
              <button type="button" className="p-1">
                {openSections.qualifications ? <ChevronUp className="w-4 h-4" /> : <ChevronDown className="w-4 h-4" />}
              </button>
            </div>

            {openSections.qualifications && (
              <div className="p-5 space-y-6">
                {/* Required Qualifications */}
                <div className="space-y-3">
                  <span className="text-xs font-black text-slate-600 dark:text-slate-300 uppercase tracking-wider">Required Qualifications</span>
                  {(jd.qualifications?.required || []).map((q, idx) => (
                    <div key={q.id || idx} className="flex items-center gap-2">
                      <input
                        type="text"
                        value={q.title}
                        onChange={(e) => updateQual('required', idx, 'title', e.target.value)}
                        placeholder="e.g. Bachelor's degree in Computer Science or related field"
                        className="flex-1 px-3 py-2 bg-white dark:bg-slate-950 border border-slate-200 dark:border-white/10 rounded-xl text-sm font-semibold outline-none focus:border-emerald-500 dark:text-white"
                      />
                      <button
                        type="button"
                        onClick={() => removeQual('required', idx)}
                        className="p-2 text-slate-400 hover:text-red-500 transition-colors"
                      >
                        <Trash2 className="w-4 h-4" />
                      </button>
                    </div>
                  ))}
                  <button
                    type="button"
                    onClick={() => addQual('required')}
                    className="py-2 px-3 text-xs font-bold text-emerald-600 dark:text-emerald-400 hover:underline flex items-center gap-1.5"
                  >
                    <Plus className="w-3.5 h-3.5" /> Add Required Qualification
                  </button>
                </div>

                {/* Preferred Qualifications */}
                <div className="space-y-3 pt-3 border-t border-slate-200 dark:border-white/10">
                  <span className="text-xs font-black text-slate-600 dark:text-slate-300 uppercase tracking-wider">Preferred Qualifications</span>
                  {(jd.qualifications?.preferred || []).map((q, idx) => (
                    <div key={q.id || idx} className="flex items-center gap-2">
                      <input
                        type="text"
                        value={q.title}
                        onChange={(e) => updateQual('preferred', idx, 'title', e.target.value)}
                        placeholder="e.g. Master's degree or AWS Certified Solutions Architect"
                        className="flex-1 px-3 py-2 bg-white dark:bg-slate-950 border border-slate-200 dark:border-white/10 rounded-xl text-sm font-semibold outline-none focus:border-emerald-500 dark:text-white"
                      />
                      <button
                        type="button"
                        onClick={() => removeQual('preferred', idx)}
                        className="p-2 text-slate-400 hover:text-red-500 transition-colors"
                      >
                        <Trash2 className="w-4 h-4" />
                      </button>
                    </div>
                  ))}
                  <button
                    type="button"
                    onClick={() => addQual('preferred')}
                    className="py-2 px-3 text-xs font-bold text-emerald-600 dark:text-emerald-400 hover:underline flex items-center gap-1.5"
                  >
                    <Plus className="w-3.5 h-3.5" /> Add Preferred Qualification
                  </button>
                </div>
              </div>
            )}
          </div>
</SortableSection>
      );
      if (sectionKey === 'eeo_statement') return (
        <SortableSection id="eeo_statement" onDelete={handleDeleteSection}>
{/* 8. EEO STATEMENT */}
          <div className="border border-slate-200 dark:border-white/10 rounded-2xl overflow-hidden bg-slate-50/50 dark:bg-white/[0.02]">
            <div
              className="w-full px-5 py-3.5 flex items-center justify-between bg-slate-100/80 dark:bg-slate-800/40 text-slate-800 dark:text-slate-200 font-bold text-sm hover:bg-slate-200/60 transition-colors cursor-pointer"
              onClick={() => toggleSection('eeo')}
            >
              <div className="flex items-center gap-2.5">
                <ShieldCheck className="w-4 h-4 text-emerald-500" />
                <span>8.</span>
                <EditableSectionTitle
                  title={(jd?.sections_metadata || {}).eeo_statement?.label || "Equal Employment Opportunity (EEO) Statement"}
                  onSave={(newTitle) => {
                    const currentMeta = jd.sections_metadata || {};
                    updateJDState(prev => ({ ...prev, sections_metadata: { ...currentMeta, eeo_statement: { ...(currentMeta.eeo_statement || {}), label: newTitle } } }), 'sections_metadata');
                  }}
                />
              </div>
              <button type="button" className="p-1">
                {openSections.eeo ? <ChevronUp className="w-4 h-4" /> : <ChevronDown className="w-4 h-4" />}
              </button>
            </div>

            {openSections.eeo && (
              <div className="p-5">
                <textarea
                  value={jd.eeo_statement || ""}
                  onChange={(e) => updateField('eeo_statement', e.target.value)}
                  placeholder="Equal Opportunity statement..."
                  rows={4}
                  className="w-full p-3.5 bg-white dark:bg-slate-950 border border-slate-200 dark:border-white/10 rounded-xl text-xs font-medium outline-none focus:border-emerald-500 dark:text-white leading-relaxed resize-y"
                />
              </div>
            )}
          </div>
</SortableSection>
      );

      // Dynamic Section Fallback
      if ([
          "id", "org_id", "creator_id", "template_id", "title", "companyName", "company_name", "jobId", "job_id", "jobFamily", "job_family", "jobLevel", "job_level", "department", "location", "city", "countryCode", "country_code", "seniority", "industry", "salary_range", "salary_symbol", "salary_min_value", "salary_max_value", "salary_period", "salary_unit", "salary_range_formatted", "employmentType", "employment_type", "key_skills", "skills", "core_competencies", "coreCompetencies", "functional_competencies", "functionalCompetencies", "additional_context", "context", "image_url", "company_logo", "content", "custom_fields", "sections_metadata", "eeoc_flags", "eeocFlags", "eeoc_cleared", "status", "public_jd_id", "wordCount", "word_count", "generation_mode", "finalized_at", "parent_jd_id", "is_main", "version_history", "created_at", "updated_at",
          "summary", "essential_duties_and_responsibilities", "essential_duties_and_responsibilities_text", "responsibilities", "qualifications", "eeo_statement", "weight_view_responsibilities_view", "weight_view_corecompetencies_view", "weight_view_functionalcompetencies_view", "weight_view_qualifications_required_view", "weight_view_qualifications_preferred_view", "basic", "salary"
        ].includes(sectionKey) || sectionKey.endsWith('_view')) {
          return null;
      }
      
      const sectionContent = jd ? jd[sectionKey] : null;
      if (sectionContent === undefined) return null;
      
      const sectionTitle = sectionKey;
      const titleStr = (jd?.sections_metadata || {})[sectionTitle]?.label || sectionTitle.replace(/_/g, ' ').toUpperCase();
      const isPoints = (jd?.sections_metadata || {})[sectionTitle]?.type === 'points';
      const sectionNumber = idx + 1;
      const isSectionOpen = openSections[sectionTitle];
      
      return (
        <SortableSection key={sectionTitle} id={sectionTitle} onDelete={handleDeleteSection}>
            <div className="border border-slate-200 dark:border-white/10 rounded-2xl overflow-hidden bg-slate-50/50 dark:bg-white/[0.02]">
              <div
                className="w-full px-5 py-3.5 flex items-center justify-between bg-slate-100/80 dark:bg-slate-800/40 text-slate-800 dark:text-slate-200 font-bold text-sm hover:bg-slate-200/60 transition-colors cursor-pointer"
                onClick={() => toggleSection(sectionTitle)}
              >
                <div className="flex items-center gap-2.5">
                  <Layers className="w-4 h-4 text-emerald-500" />
                  <span>{sectionNumber}.</span>
                  <EditableSectionTitle
                    title={titleStr}
                    onSave={(newTitle) => {
                      const currentMeta = jd.sections_metadata || {};
                      updateJDState(prev => ({
                        ...prev,
                        sections_metadata: {
                          ...currentMeta,
                          [sectionTitle]: {
                            ...(currentMeta[sectionTitle] || {}),
                            label: newTitle
                          }
                        }
                      }), 'sections_metadata');
                    }}
                  />
                </div>
                <button type="button" className="p-1">
                  {isSectionOpen ? <ChevronUp className="w-4 h-4 text-slate-400" /> : <ChevronDown className="w-4 h-4 text-slate-400" />}
                </button>
              </div>

              {isSectionOpen && (
                <div className="p-5">
                  {isPoints || Array.isArray(sectionContent) ? (
                    <div className="space-y-2">
                      {(Array.isArray(sectionContent) ? sectionContent : [sectionContent]).map((point, pIdx) => (
                        <div key={pIdx} className="flex items-start gap-2">
                          <span className="text-slate-400 font-bold mt-1.5">•</span>
                          <input
                            type="text"
                            value={typeof point === 'string' ? point.replace(/^[•-]\s*/, '') : (point?.point || point?.title || "")}
                            onChange={(e) => {
                              const currentArray = [...(Array.isArray(sectionContent) ? sectionContent : [sectionContent])];
                              if (typeof currentArray[pIdx] === 'object') {
                                currentArray[pIdx] = { ...currentArray[pIdx], point: e.target.value, title: e.target.value };
                              } else {
                                currentArray[pIdx] = e.target.value;
                              }
                              updateJDState(prev => ({ ...prev, [sectionTitle]: currentArray }), sectionTitle);
                            }}
                            placeholder={`Point ${pIdx + 1}...`}
                            className="flex-1 px-3 py-1.5 bg-white dark:bg-slate-950 border border-slate-200 dark:border-white/10 rounded-lg text-xs font-medium outline-none focus:border-emerald-500 dark:text-white"
                          />
                          <button
                            type="button"
                            onClick={() => {
                              const currentArray = [...(Array.isArray(sectionContent) ? sectionContent : [sectionContent])];
                              currentArray.splice(pIdx, 1);
                              if (currentArray.length === 0) currentArray.push("");
                              updateJDState(prev => ({ ...prev, [sectionTitle]: currentArray }), sectionTitle);
                            }}
                            className="p-1.5 text-slate-400 hover:text-red-500 transition-colors mt-0.5"
                          >
                            <Trash2 className="w-3.5 h-3.5" />
                          </button>
                        </div>
                      ))}
                      <button
                        type="button"
                        onClick={() => {
                          const currentArray = [...(Array.isArray(sectionContent) ? sectionContent : [sectionContent])];
                          currentArray.push("");
                          updateJDState(prev => ({ ...prev, [sectionTitle]: currentArray }), sectionTitle);
                        }}
                        className="text-[10px] font-bold text-emerald-600 dark:text-emerald-400 hover:underline flex items-center gap-1 mt-1 ml-4"
                      >
                        <Plus className="w-3.5 h-3.5" /> Add Point
                      </button>
                    </div>
                  ) : (
                    <textarea
                      value={sectionContent || ""}
                      onChange={(e) => updateJDState(prev => ({ ...prev, [sectionTitle]: e.target.value }), sectionTitle)}
                      placeholder={`Enter ${titleStr}...`}
                      rows={3}
                      className="w-full px-3 py-2 bg-white dark:bg-slate-950 border border-slate-200 dark:border-white/10 rounded-lg text-xs font-medium outline-none focus:border-emerald-500 dark:text-white resize-y"
                    />
                  )}
                </div>
              )}
            </div>
            </SortableSection>
      );
    })}
  </SortableContext>
</DndContext>
